from nlpoptnet import NLPOptNet

CONFIG = {

    "hidden_size": 32,
    "hidden_layers": 2,

    "dtype": "float64",
    "print_every": 1000,
    "verbose": True,
    # Configurations from config.json
    # Configurations from proj.json

param_path = "parameters.csv" (If there's parameters available)
problem_path = "problem.npz" (If the problem is structured with the matrices saved)

Create the problem:

model = NLPOptNet(config=CONFIG) (For general) or model = NLPOptNet(config, type="QP"/"QCQP"/"NLP")

### Defining Variables and Parameters ###
x = model.add_parameter(["x1", "x2"])
y = model.add_variable(["y1", "y2", "y3"])

### Extracting the Matrices if available ###
# This should extract and print the matrices or vectors you have in problem.npz. 
Here the matrices and vectors you have in problem.npz will become instances of the model. So if there is a matrix like Q in problem.npz you can acccess it like model.Q

model.extract(path=problem_path)

or you can define matrix and vectors here like
model.Q = model.matrix([[1, 2], [2, 1]])
model.c = model.vector([1, 2])

### Defining Objective ###
model.objective(0.5 * (y.y1**2 + y.y2**2 + y.y3**2))
or
model.objective(0.5 * model.quad(model.Q, y) (Quadratic Term => yTQy) + model.lin(model.c, y) (Linear Term => cTy)) # Quadratic objective
or
model.objective(0.5 * model.quad(model.Q, y) + model.lin(model.p, model.sin(y))) # Nonlinear Objective (similarly there can be other functions also)
or you can define your objective function

def obj():
    quadratic = 0.5 * y @ (model.Q @ y)
    sinusoidal = model.c @ jnp.sin(y)
    return quadratic + sinusoidal

model.objective(obj)

# model.quad and model.lin is functions for quadratic and linear operations. model.quad(Q, y) means yTQy and model.lin(c, y) means cTy
###  Adding the Constraints ###

model.constraints.equality.add(

model.constraints.inequality.add(
    model.batch_quad(model.C, y) + model.batch_lin(model.d, y) <= model.beta + model.batch_lin(model.E, x) # Quadratic Inequality Constraints
)

model.constraints.box.add(
    model.l + model.lin(model.L, x) <= y <= model.u + model.lin(model.U, x) # Bound Constraints
)

# model.batch_quad and model.batch_lin means batch quadratic and batch linear operations. For example,

this is the actual formulation of the written qcqp problem:
\begin{ceqn}
\begin{equation}\label{qcqp_form}
\begin{aligned}
\min_{y} \quad & \frac{1}{2} y^\top Q y + c^\top y \\
\text{s.t.} \quad 
& A y = b + B x, \\
& y^\top C_i y + d_i^\top y \leq \beta_i + E_i x, \quad i = 1, 2, \ldots, m_i, \\
& l + L x \leq y \leq u + U x.
\end{aligned}
\end{equation}
\end{ceqn}

# model.sin, model.exp should all be jnp functions
or

# NLP Case

model.constraints.equality.add(
    model.lin(model.A, y) == model.b + model.lin(model.B, x), # Equality Constraints
)

model.constraints.inequality.add(
    model.batch_lin(model.a, model.batch_exp(y)) + model.batch_quad(model.W, y) <= model.beta + model.batch_lin(model.E, x) # Nonlinear Inequality Constraints
)

model.constraints.box.add(
    model.l + model.lin(model.L, x) <= y <= model.u + model.lin(model.U, x) # Bound Constraints
)

# this is the actual formulation of the written nlp problem:
\begin{ceqn}
\begin{equation}\label{nlp_form}
\begin{aligned}
\min_{y} \quad & \frac{1}{2} y^\top Q y + c^\top y \\
\text{s.t.} \quad 
& A y = b + B x, \\
& a_i^\top \exp(y) + y^\top W y \leq \beta_i + E_i x, \quad i = 1, 2, \ldots, m_i, \\
& l + L x \leq y \leq u + U x.
\end{aligned}
\end{equation}
\end{ceqn}

or you can pass functions inside the add function


def equality_block(params, vars):
    x = params[PARAM_NAME]
    y = vars["y"]
    quad = jnp.einsum("mij,i,j->m", model.A, y, y)
    linear = model.c @ y
    return quad + linear + model.d - x[: model.A.shape[0]] ** 3


def inequality_block(params, vars):
    del params
    y = vars["y"]
    quad = jnp.einsum("mij,i,j->m", model.H, y, y)
    linear = model.g @ y
    return quad + linear - model.r

model.constraints.equality.add(
    equality_block # Equality Constraints
)

model.constraints.inequality.add(
    inequality_block # Nonlinear Inequality Constraints
)

model.constraints.box.add(
    model.l + model.lin(model.L, x) <= y <= model.u + model.lin(model.U, x) # Bound Constraints
)
